# Bloque a reemplazar en app.py (sql_recs_dg_hybrid)
# --- 7, 8, 9. Recomendaciones (Híbrido) y Total Global (MODIFICADO PARA FILTRAR POR AÑO) ---
sql_recs_dg_hybrid = """
    WITH DGs AS ( SELECT id_dg, siglas_dg FROM dgs WHERE activo = TRUE ), 
    RealRecsCounts AS ( 
        SELECT 
            d.id_dg, 
            COUNT(rec.id_recomendacion) as real_emitidas_count, 
            COUNT(CASE WHEN rec.estatus IN ('Completada', 'Cerrada') THEN 1 END) as real_atendidas_count 
        FROM DGs d 
        LEFT JOIN ramos r ON d.id_dg = r.id_dg 
        LEFT JOIN instituciones i ON r.id_ramo = i.id_ramo AND i.activo = TRUE 
        LEFT JOIN recomendaciones rec ON i.id_institucion = rec.id_institucion AND rec.activo = TRUE 
        -- FILTRO CRÍTICO: Solo contar recomendaciones emitidas hasta el año seleccionado
        WHERE EXTRACT(YEAR FROM rec.fecha_emision)::INT <= %s OR rec.fecha_emision IS NULL 
        GROUP BY d.id_dg 
    ), 
    HistoricInformesSums AS ( 
        SELECT 
            d.id_dg, 
            COALESCE(SUM(inf.recomendaciones_emitidas), 0) as historic_emitidas_sum, 
            COALESCE(SUM(inf.recomendaciones_atendidas), 0) as historic_atendidas_sum 
        FROM DGs d 
        LEFT JOIN ramos r ON d.id_dg = r.id_ramo 
        LEFT JOIN instituciones i ON r.id_ramo = i.id_ramo AND i.activo = TRUE 
        LEFT JOIN informes inf ON i.id_institucion = inf.id_institucion AND inf.activo = TRUE 
        -- FILTRO CRÍTICO: Solo contar informes reportados hasta el año seleccionado
        WHERE inf.periodo_reportado::INT <= %s 
          AND NOT EXISTS ( SELECT 1 FROM recomendaciones r_check WHERE r_check.id_informe = inf.id_informe AND r_check.activo = TRUE ) 
        GROUP BY d.id_dg 
    ) 
    SELECT 
        d.id_dg, d.siglas_dg, 
        (COALESCE(rr.real_emitidas_count, 0) + COALESCE(hi.historic_emitidas_sum, 0)) as emitidas_count, 
        (COALESCE(rr.real_atendidas_count, 0) + COALESCE(hi.historic_atendidas_sum, 0)) as atendidas_count, 
        CASE WHEN (COALESCE(rr.real_emitidas_count, 0) + COALESCE(hi.historic_emitidas_sum, 0)) > 0 
             THEN ROUND( ( (COALESCE(rr.real_atendidas_count, 0) + COALESCE(hi.historic_atendidas_sum, 0))::DECIMAL / (COALESCE(rr.real_emitidas_count, 0) + COALESCE(hi.historic_emitidas_sum, 0)) ) * 100, 2) 
             ELSE 0 
        END as efficiency_pct 
    FROM DGs d 
    LEFT JOIN RealRecsCounts rr ON d.id_dg = rr.id_dg 
    LEFT JOIN HistoricInformesSums hi ON d.id_dg = hi.id_dg 
    ORDER BY d.siglas_dg;
"""

# Bloque a reemplazar en app.py (Línea de ejecución de sql_recs_dg_hybrid - aprox. Línea 795)
# Reemplazar: cur.execute(sql_recs_dg_hybrid)
